cmake_minimum_required(VERSION 3.25)
project(AxiomChess VERSION 1.0.0 LANGUAGES C CXX)
if(MSVC)
  # Search cancellation uses C++ exceptions and scope guards.
  add_compile_options(/EHsc /utf-8)
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(AXIOM_SANITIZE "Enable address/undefined sanitizers on GCC/Clang" OFF)
option(AXIOM_NATIVE_AVX2 "Build for CPUs supporting AVX2" OFF)
option(AXIOM_RESEARCH_TRACE "Compile bounded causal trace instrumentation (research only)" OFF)
set(FATHOM_DIR "" CACHE PATH "Optional Fathom checkout containing src/tbprobe.c")
add_library(axiom_core src/chess.cpp src/evaluation.cpp src/strategic.cpp src/search.cpp src/qsearch.cpp src/move_order.cpp src/search_report.cpp src/feedback.cpp src/proof.cpp src/tablebase.cpp src/retrograde.cpp src/bench.cpp)
target_include_directories(axiom_core PUBLIC include)
if(AXIOM_RESEARCH_TRACE)
  target_compile_definitions(axiom_core PUBLIC AXIOM_RESEARCH_TRACE=1)
endif()
if(MSVC)
  if(AXIOM_SANITIZE)
    # Use RelWithDebInfo: Debug /RTC is incompatible with AddressSanitizer.
    target_compile_options(axiom_core PUBLIC /fsanitize=address /Zi)
    target_link_options(axiom_core PUBLIC /INCREMENTAL:NO)
  endif()
  target_compile_options(axiom_core PRIVATE /W4 /permissive- /utf-8)
  target_compile_options(axiom_core PRIVATE "$<$<CONFIG:Release>:/O2>" "$<$<CONFIG:Release>:/Ob3>" "$<$<CONFIG:Release>:/Oi>")
  if(AXIOM_NATIVE_AVX2)
    target_compile_options(axiom_core PRIVATE /arch:AVX2)
    target_compile_definitions(axiom_core PRIVATE AXIOM_SIMD_AVX2=1)
  endif()
else()
  target_compile_options(axiom_core PRIVATE -Wall -Wextra -Wpedantic)
  if(AXIOM_NATIVE_AVX2)
    target_compile_options(axiom_core PRIVATE -mavx2 -mbmi2 -mpopcnt)
    target_compile_definitions(axiom_core PRIVATE AXIOM_SIMD_AVX2=1)
  endif()
  if(AXIOM_SANITIZE)
    target_compile_options(axiom_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(axiom_core PUBLIC -fsanitize=address,undefined)
  endif()
endif()
if(FATHOM_DIR)
  add_library(fathom STATIC "${FATHOM_DIR}/src/tbprobe.c")
  target_include_directories(fathom PUBLIC "${FATHOM_DIR}/src")
  target_link_libraries(axiom_core PRIVATE fathom)
  target_compile_definitions(axiom_core PRIVATE AXIOM_FATHOM=1)
endif()
find_package(Threads REQUIRED)
target_link_libraries(axiom_core PUBLIC Threads::Threads)
target_sources(axiom_core PRIVATE src/tt.cpp)
add_executable(axiom src/main.cpp src/protocol.cpp)
set_target_properties(axiom PROPERTIES OUTPUT_NAME "axiom-1.0")
if(AXIOM_RESEARCH_TRACE)
  set_target_properties(axiom PROPERTIES OUTPUT_NAME "axiom-1.0-research")
endif()
target_link_libraries(axiom PRIVATE axiom_core Threads::Threads)
add_executable(axiom_tests tests/tests.cpp)
target_link_libraries(axiom_tests PRIVATE axiom_core)
add_executable(axiom_corpus src/corpus.cpp)
target_link_libraries(axiom_corpus PRIVATE axiom_core)
set_property(TARGET axiom_core axiom axiom_tests axiom_corpus PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
enable_testing()
add_test(NAME axiom_tests COMMAND axiom_tests)
add_executable(axiom_audit tests/audit.cpp)
target_link_libraries(axiom_audit PRIVATE axiom_core)
set_property(TARGET axiom_audit PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
add_test(NAME axiom_audit COMMAND axiom_audit)
add_executable(axiom_calibration tests/calibration.cpp)
target_link_libraries(axiom_calibration PRIVATE axiom_core)
set_property(TARGET axiom_calibration PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
add_test(NAME axiom_calibration COMMAND axiom_calibration)
add_executable(axiom_pgn src/pgn.cpp)
target_link_libraries(axiom_pgn PRIVATE axiom_core)
set_property(TARGET axiom_pgn PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
set_property(DIRECTORY PROPERTY VS_STARTUP_PROJECT axiom)
add_executable(axiom_holdout src/holdout.cpp)
target_link_libraries(axiom_holdout PRIVATE axiom_core)
add_executable(axiom_legal_fast tests/legal_fast.cpp)
target_link_libraries(axiom_legal_fast PRIVATE axiom_core)
add_test(NAME axiom_legal_fast COMMAND axiom_legal_fast)

# Independent Axiom 2 implementation; the legacy engine remains an oracle.
add_library(axiom2_core v2/attacks.cpp v2/position.cpp v2/movegen.cpp v2/evaluation.cpp v2/see.cpp)
target_include_directories(axiom2_core PUBLIC v2)
target_link_libraries(axiom2_core PUBLIC Threads::Threads)
if(MSVC)
  target_compile_options(axiom2_core PRIVATE /W4 /permissive- $<$<CONFIG:Release>:/O2>)
  if(AXIOM_SANITIZE)
    target_compile_options(axiom2_core PUBLIC /fsanitize=address /Zi)
    target_link_options(axiom2_core PUBLIC /INCREMENTAL:NO)
  endif()
  if(AXIOM_NATIVE_AVX2)
    target_compile_options(axiom2_core PRIVATE /arch:AVX2)
  endif()
else()
  target_compile_options(axiom2_core PRIVATE -Wall -Wextra -Wpedantic)
  if(AXIOM_NATIVE_AVX2)
    target_compile_options(axiom2_core PRIVATE -mavx2)
  endif()
  if(AXIOM_SANITIZE)
    target_compile_options(axiom2_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(axiom2_core PUBLIC -fsanitize=address,undefined)
  endif()
endif()
add_executable(axiom2_core_tests v2/core_tests.cpp)
target_link_libraries(axiom2_core_tests PRIVATE axiom2_core axiom_core)
add_test(NAME axiom2_core_tests COMMAND axiom2_core_tests)
target_sources(axiom2_core PRIVATE v2/tt.cpp v2/time.cpp v2/ordering.cpp v2/qsearch.cpp v2/search.cpp v2/root.cpp v2/stats.cpp v2/threads.cpp v2/tablebase.cpp)
target_sources(axiom2_core PRIVATE v2/bench.cpp)
add_executable(axiom2 v2/uci.cpp)
target_link_libraries(axiom2 PRIVATE axiom2_core)
set_target_properties(axiom2 PROPERTIES OUTPUT_NAME "axiom-2")
add_executable(axiom2_search_tests v2/search_tests.cpp)
target_link_libraries(axiom2_search_tests PRIVATE axiom2_core)
add_test(NAME axiom2_search_tests COMMAND axiom2_search_tests)
add_executable(axiom2_openings v2/openings.cpp)
target_link_libraries(axiom2_openings PRIVATE axiom2_core)
set_property(TARGET axiom2_core axiom2 axiom2_core_tests axiom2_search_tests PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
if(FATHOM_DIR)
  target_link_libraries(axiom2_core PRIVATE fathom)
  target_compile_definitions(axiom2_core PRIVATE AXIOM_FATHOM=1)
endif()
